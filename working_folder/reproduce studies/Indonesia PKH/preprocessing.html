Indonesia PKH: preprocessing

Data preparation

Link the supplied 2013 maternal records to pregnancy histories using iid_w4. The authors’ coded fertility dataset already contains the household treatment, district, subdistrict, and baseline controls used in their maternal analysis. This avoids rebuilding the full Stata pipeline. Raw files remain unchanged.

Code 
suppressPackageStartupMessages({
  library(haven)
  library(dplyr)
})

dat <- local({
  coded_file <- "data/data/coded/mothers_fertility_timing.dta"
  metadata <- read_dta(coded_file, n_max = 0)
  controls <- names(metadata)[grepl("_baseline_(nm|miss)$", names(metadata))]
  # table2.do excludes the food/non-food expenditure controls and their flags.
  controls <- controls[!grepl("logpcexp_(food|nonfood)", controls)]
  mothers <- read_dta(coded_file, col_select = all_of(c(
    "iid", "iid_w4", "rid_merge", "survey_round", "L07", "pkh_by_this_wave",
    "kabupaten", "kecamatan", controls
  ))) |>
    filter(survey_round == 2, !is.na(L07))
  stopifnot(!anyDuplicated(mothers$iid_w4))

  # Undo the authors' zero filling using the source missingness flags.
  value_columns <- grep("_baseline_nm$", controls, value = TRUE)
  for (variable in value_columns) {
    missing <- mothers[[sub("_nm$", "_miss", variable)]]
    value <- as.numeric(mothers[[variable]])
    value[is.na(missing) | missing == 1] <- NA_real_
    mothers[[variable]] <- value
  }

  # Each complete category block contains exactly one active dummy.
  category_columns <- character()
  for (prefix in c("hh_educ_", "roof_type", "wall_type", "floor_type")) {
    columns <- grep(paste0("^", prefix, "[0-9]+_baseline_nm$"),
                    value_columns, value = TRUE)
    categories <- as.integer(sub("_baseline_nm$", "", sub(prefix, "", columns)))
    columns <- columns[order(categories)]
    categories <- sort(categories)
    block <- as.matrix(mothers[columns])
    observed <- complete.cases(block)
    stopifnot(all(block[observed, , drop = FALSE] %in% 0:1),
              all(rowSums(block[observed, , drop = FALSE]) == 1))
    category <- rep(NA_integer_, nrow(mothers))
    category[observed] <- categories[max.col(block[observed, , drop = FALSE])]
    mothers[[paste0(sub("_$", "", prefix), "_baseline")]] <-
      factor(category, levels = categories)
    category_columns <- c(category_columns, columns)
  }
  for (variable in setdiff(value_columns, category_columns)) {
    mothers[[sub("_nm$", "", variable)]] <- mothers[[variable]]
  }
  mothers <- mothers |> select(-all_of(controls))

  pregnancies <- read_dta(
    "data/data/raw/Wave IV/I_CH.dta",
    col_select = c("iid", "ich02_line", "ich02", "ich24", "ich25")
  )
  out <- mothers |>
    inner_join(pregnancies, by = c("iid_w4" = "iid"), relationship = "one-to-many") |>
    mutate(
      L07 = as.integer(L07),
      pkh_by_this_wave = as.integer(pkh_by_this_wave),
      kabupaten = factor(as.integer(kabupaten)),
      kecamatan = as.integer(kecamatan),
      good_assisted_delivery = if_else(
        is.na(ich25) | trimws(ich25) == "", NA_integer_,
        as.integer(grepl("A|B", ich25))
      ),
      delivery_facility = if_else(is.na(ich24), NA_integer_, as.integer(ich24 < 7))
    )
  stopifnot(!anyDuplicated(out[c("iid_w4", "ich02_line")]),
            all(out$L07 %in% 0:1), all(out$pkh_by_this_wave %in% c(0, 1, NA)),
            all(out$good_assisted_delivery %in% c(0, 1, NA)),
            all(out$delivery_facility %in% c(0, 1, NA)))
  out
})
save(dat, file = "processed_data.rda", compress = "xz")

The unit is a pregnancy record, so a mother can contribute more than one record. Outcomes are observed for deliveries; ongoing pregnancies and other inapplicable records can have missing outcomes. The analysis uses a common sample with observed outcomes, treatment, instrument, and geography, and handles missing baseline predictors during model preparation.

good_assisted_delivery follows the authors’ ich25 coding: doctor (A) or midwife (B). delivery_facility follows ich24 < 7; missing responses remain missing. The treatment is ever received PKH by the 2013 wave, not just current receipt. Baseline education, roof, wall, and floor variables are unordered factors; their levels retain the authors’ dummy category indices, not questionnaire codes. Other baseline controls remain numeric. Source missingness is restored to NA, and the old dummy columns and missingness flags are removed from the saved data.

Variables and availability

Missing percentages describe the linked records before analysis exclusions and include the restored baseline missing values.

Code 
local({
  labels <- c(
    iid = "Woman identifier aligned across survey waves", iid_w4 = "2013 woman identifier",
    rid_merge = "Household linkage identifier", survey_round = "Survey round; 2 denotes 2013",
    L07 = "Randomized subdistrict rollout in 2007",
    pkh_by_this_wave = "Household ever received PKH by this wave",
    kabupaten = "District", kecamatan = "Subdistrict; randomization and clustering unit",
    ich02_line = "Pregnancy record number", ich02 = "Pregnancy status/type",
    ich24 = "Delivery location code", ich25 = "Delivery-attendant response",
    good_assisted_delivery = "Delivery assisted by a doctor or midwife",
    delivery_facility = "Delivery at a health facility, authors' coding",
    hh_educ_baseline = "Baseline household-head education category",
    roof_type_baseline = "Baseline roof-material category",
    wall_type_baseline = "Baseline wall-material category",
    floor_type_baseline = "Baseline floor-material category",
    hh_head_agr_baseline = "Baseline household head in agriculture",
    hh_head_serv_baseline = "Baseline household head in services",
    clean_water_baseline = "Baseline clean water",
    own_latrine_baseline = "Baseline own latrine",
    square_latrine_baseline = "Baseline latrine type indicator",
    own_septic_tank_baseline = "Baseline own septic tank",
    electricity_PLN_baseline = "Baseline PLN electricity connection",
    logpcexp_baseline = "Baseline log per-capita expenditure",
    hhsize_ln_baseline = "Baseline log household size"
  )
  variables <- tibble(Variable = names(dat)) |>
    mutate(
      Source = ifelse(Variable %in% c("ich02_line", "ich02", "ich24", "ich25",
                                      "good_assisted_delivery", "delivery_facility"),
                      "Wave IV/I_CH.dta", "mothers_fertility_timing.dta"),
      Role = case_when(
        Variable %in% c("good_assisted_delivery", "delivery_facility") ~ "Outcome",
        Variable == "L07" ~ "Instrument",
        Variable == "pkh_by_this_wave" ~ "Treatment",
        Variable == "kabupaten" ~ "District adjustment",
        Variable == "kecamatan" ~ "Cluster identifier",
        grepl("_baseline$", Variable) ~ "Baseline control",
        TRUE ~ "Linkage, selection, or source field"
      ),
      Description = unname(labels[Variable]),
      Type = vapply(dat, function(x) {
        if (is.factor(x)) return(paste("Factor:", nlevels(x), "levels"))
        if (is.numeric(x) && all(na.omit(x) %in% 0:1)) return("Binary")
        class(x)[1]
      }, character(1)),
      Availability = "Available where recorded/applicable; missing values are NA",
      Missing_percent = vapply(dat, function(x) round(100 * mean(is.na(x)), 2), numeric(1))
    )
  knitr::kable(variables, row.names = FALSE)
})
